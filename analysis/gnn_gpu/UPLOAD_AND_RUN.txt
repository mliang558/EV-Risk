### GAT+Global — 只预测 random（10 维）

  --dataset pyg/pyg_dataset.pt --model gat --global-features \
  --attack random --out pyg/train_gat_global_random

### 评估两个单任务 ckpt
PYTHONPATH=. $PY -u step7_evaluate_gnn.py --dataset pyg/pyg_dataset.pt \
  --checkpoint pyg/train_gat_global_betweenness/best_model.pt \
  --out results_gnn/eval_gat_global_betweenness --no-plots

PYTHONPATH=. $PY -u step7_evaluate_gnn.py --dataset pyg/pyg_dataset.pt \
  --checkpoint pyg/train_gat_global_random/best_model.pt \
  --out results_gnn/eval_gat_global_random --no-plots

### 对比 multi-task vs single-task（拼 B/R 两列 R²）
PYTHONPATH=. $PY -u step7_compare_single_vs_multitask.py --data-dir . \
  --multitask pyg/train_gat_global_br/best_model.pt \
  --betweenness pyg/train_gat_global_betweenness/best_model.pt \
  --random pyg/train_gat_global_random/best_model.pt \
  --eval-multitask results_gnn/eval_gat_global_br/test_metrics.json \
  --out results_gnn/single_vs_multitask
